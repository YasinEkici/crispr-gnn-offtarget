\newcommand{\Coord}[2]{([xshift=#1cm,yshift=-#2cm]current page.north west)}

\newcommand{\PosterBlock}[7][\centering]{%
  % #1 hizalama komutu: \centering, \raggedright, \justifying vb.
  % #2 x, #3 y, #4 genişlik, #5 gövde yüksekliği, #6 başlık, #7 içerik
  \pgfmathsetmacro{\rightX}{#2 + #4}%
  \pgfmathsetmacro{\midX}{#2 + #4/2}%
  \pgfmathsetmacro{\barBottomY}{#3 + \TitleBarHeight}%
  \pgfmathsetmacro{\barMidY}{#3 + \TitleBarHeight/2}%
  \pgfmathsetmacro{\bodyBottomY}{\barBottomY + #5}%
  \pgfmathsetmacro{\bodyX}{#2 + 0.55}%
  \pgfmathsetmacro{\bodyY}{\barBottomY + 0.45}%
  \pgfmathsetmacro{\bodyW}{#4 - 1.10}%
  \pgfmathsetmacro{\bodyH}{#5 - 0.80}%
  \fill[BTUBlue] \Coord{#2}{#3} rectangle \Coord{\rightX}{\barBottomY};
  \node[anchor=center,text=white,align=center] at \Coord{\midX}{\barMidY} {\BlockTitleFont #6};
  \draw[black,line width=0.55pt] \Coord{#2}{\barBottomY} rectangle \Coord{\rightX}{\bodyBottomY};
  \node[anchor=north west,inner sep=0pt] at \Coord{\bodyX}{\bodyY} {%
    \begin{minipage}[t][\bodyH cm][t]{\bodyW cm}
      #1\BodyFont
      #7
    \end{minipage}%
  };
}

\newcommand{\FigureSlot}[6]{%
  % #1 x, #2 y, #3 genişlik, #4 yükseklik, #5 şekil no, #6 açıklama
  \pgfmathsetmacro{\rightX}{#1 + #3}%
  \pgfmathsetmacro{\bottomY}{#2 + #4}%
  \pgfmathsetmacro{\midX}{#1 + #3/2}%
  \pgfmathsetmacro{\midY}{#2 + #4/2}%
  \pgfmathsetmacro{\capY}{#2 + #4 + 0.30}%
  \fill[PlaceholderGray] \Coord{#1}{#2} rectangle \Coord{\rightX}{\bottomY};
  \draw[PlaceholderLine,line width=0.55pt] \Coord{#1}{#2} rectangle \Coord{\rightX}{\bottomY};
  \node[anchor=center,align=center,text=black!45] at \Coord{\midX}{\midY} {\TinyPlaceholderFont Şekil / grafik alanı};
  \node[anchor=north west,inner sep=0pt,text width=#3 cm,align=center] at \Coord{#1}{\capY} {\CaptionFont\textbf{Şekil #5.} #6};
}

\newcommand{\ImageWithCaption}[7]{%
  % #1 x, #2 y, #3 genişlik, #4 yükseklik, #5 dosya yolu, #6 şekil no, #7 açıklama
  \node[anchor=north west,inner sep=0pt] at \Coord{#1}{#2} {\includegraphics[width=#3 cm,height=#4 cm,keepaspectratio]{#5}};
  \pgfmathsetmacro{\capY}{#2 + #4 + 0.30}%
  \node[anchor=north west,inner sep=0pt,text width=#3 cm,align=center] at \Coord{#1}{\capY} {\CaptionFont\textbf{Şekil #6.} #7};
}

\newcommand{\TableCaption}[5]{%
  % #1 x, #2 y, #3 genişlik, #4 tablo no, #5 açıklama
  \node[anchor=north west,inner sep=0pt,text width=#3 cm,align=center] at \Coord{#1}{#2} {\CaptionFont\textbf{Tablo #4.} #5};
}

\newcommand{\TablePlaceholder}[4]{%
  % #1 x, #2 y, #3 genişlik, #4 yükseklik
  \pgfmathsetmacro{\rightX}{#1 + #3}%
  \pgfmathsetmacro{\bottomY}{#2 + #4}%
  \pgfmathsetmacro{\rowH}{#4/6}%
  \pgfmathsetmacro{\colW}{#3/4}%
  \pgfmathsetmacro{\headerY}{#2 + \rowH}%
  \fill[BTUBlue] \Coord{#1}{#2} rectangle \Coord{\rightX}{\headerY};
  \draw[black!45,line width=0.55pt] \Coord{#1}{#2} rectangle \Coord{\rightX}{\bottomY};
  \foreach \i in {1,2,3} {\pgfmathsetmacro{\xline}{#1 + \i*\colW}\draw[black!45,line width=0.45pt] \Coord{\xline}{#2} -- \Coord{\xline}{\bottomY};}
  \foreach \j in {1,2,3,4,5} {\pgfmathsetmacro{\yline}{#2 + \j*\rowH}\draw[black!45,line width=0.45pt] \Coord{#1}{\yline} -- \Coord{\rightX}{\yline};}
  \pgfmathsetmacro{\headMidY}{#2 + 0.5*\rowH}%
  \foreach \c/\txt in {1/Başlık,2/Başlık,3/Başlık} {\pgfmathsetmacro{\xmid}{#1 + (\c + 0.5)*\colW}\node[anchor=center,text=white] at \Coord{\xmid}{\headMidY} {\BodyFont \txt};}
  \foreach \r in {1,...,5} {\pgfmathsetmacro{\ymid}{#2 + (\r+0.5)*\rowH}\pgfmathsetmacro{\xmid}{#1 + 0.5*\colW}\node[anchor=center,text=black] at \Coord{\xmid}{\ymid} {\BodyFont Değer};}
}
