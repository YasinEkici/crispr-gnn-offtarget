  % #1 alignment, #2 x, #3 y, #4 width, #5 body height, #6 title, #7 content

  \pgfmathsetmacro{\barBottomY}{#3 + \TitleBarHeight}%
  \pgfmathsetmacro{\barMidY}{#3 + \TitleBarHeight/2}%

  \pgfmathsetmacro{\bodyY}{\barBottomY + 0.45}%

  \pgfmathsetmacro{\bodyH}{#5 - 0.80}%

  \node[anchor=center,text=white,align=center] at \Coord{\midX}{\barMidY} {\BlockTitleFont #6};

  };
}
